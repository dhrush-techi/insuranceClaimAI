// FRONTEND
cd frontend
npm install
npm run dev

// BACKEND (FIRST TIME SETUP)
cd backend
python -m venv venv
venv\Scripts\activate
python -m pip install --upgrade pip setuptools wheel
python -m pip install -r requirements.txt

// BACKEND (RUN API SERVER)
cd backend
venv\Scripts\activate
python -m uvicorn app:app --host 0.0.0.0 --port 8000 --reload

// RESEARCH BENCHMARK EVALUATION (RUN EXPERIMENTS)
cd backend
venv\Scripts\activate
python -m evaluation.runner

// RESEARCH DASHBOARD (VIEW RESULTS)
Open research_results/index.html in browser or serve via static HTTP server
